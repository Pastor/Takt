# Реестр НЕПОКРЫТЫХ конструкций языка — УЗАКОНЕННЫЙ ДОЛГ (фича 0403).
#
# Правило: каждая конструкция языка обязана быть показана примером — и в
# `examples/` (исполняемая документация, её гоняют гейты предкоммита), и в
# `book/src/**` (справочник языка, правило 24). Перечисленные ниже правило
# НАРУШАЮТ. Реестр не даёт им разрешения — он их ЗАМОРАЖИВАЕТ: сторож
# `takt-lang/tests/syntax/language_coverage_tests.rs` запрещает появление новых
# непокрытых конструкций, а запись о конструкции, для которой пример появился,
# объявляет протухшей и роняет прогон.
#
# Единственная допустимая правка записи — УДАЛЕНИЕ: покрытие может только
# расти. Формат: <область> <вид конструкции> # <причина>
#
# Области: `examples` и `book` — долг соответствующего корпуса; `never` —
# конструкция в каноничном примере появиться НЕ МОЖЕТ, и это доказано, а не
# заявлено (причина у каждой записи своя).

# ── Не может быть показана примером ───────────────────────────────────────
never FormulaStatement::Error # узел восстановления после ошибки в блоке `formula`: тот же случай, что `Statement::Error`
never ModelElement::StraySemicolon # лишнюю `;` стирает `taktc fmt` (проба 2026-08-23), а канон примеров сторожит `fmt --check`
never StateElement::StraySemicolon # лишнюю `;` в теле состояния стирает `taktc fmt` — та же причина
never Statement::Error # узел восстановления после ошибки: живёт только на НЕВЕРНОМ входе, каноничный пример его не даст

# ── Долг корпуса `examples/` ──────────────────────────────────────────────
examples Expression::List # список в скобках `(a, b)` разбирается, но семантики и потребителя у него нет — примером учить нечему
examples Expression::String # строкового ТИПА в языке нет (лексема `string` мертва, класс 0201): записи со смыслом для примера не существует

# ── Заблокировано форматтером ─────────────────────────────────────────────
#
# `taktc fmt` печать узлов `formula` и `assembly` не поддерживает (`FM-001`,
# замер 2026-08-23 — прогон на живом примере), а канон корпуса сторожит
# `fmt --check`: пример с этими конструкциями в `examples/` не удержится,
# пока печать не заведена. Это ПРОБЕЛ ИНСТРУМЕНТА, найденный самой фичей
# 0403, а не свойство языка: обе конструкции разбираются и переводятся.
examples ModelElement::Formula # `taktc fmt` не печатает узел `formula` (FM-001) — пример не пройдёт `fmt --check`
examples Statement::Formula # блок `formula` в теле: та же причина
examples Statement::Assembly # `taktc fmt` не печатает узел `assembly` (FM-001) — та же причина
examples FormulaStatement::Block # достижим только внутри блока `formula`
examples FormulaStatement::Function # достижим только внутри блока `formula`
examples FormulaExpression::Bool # достижим только внутри блока `formula`
examples FormulaExpression::Function # достижим только внутри блока `formula`
examples FormulaExpression::Number # достижим только внутри блока `formula`
examples FormulaExpression::String # достижим только внутри блока `formula`
examples FormulaExpression::SuffixAccess # достижим только внутри блока `formula`
examples FormulaExpression::Variable # достижим только внутри блока `formula`
